cmake_minimum_required(VERSION 3.16)
project(balloon_recomp CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
# The recompiler's runtime (CPU, DOS/BIOS, EGA, timer, speaker)
set(RC_RUNTIME ${CMAKE_CURRENT_SOURCE_DIR}/runtime)
# Generated from the shareware BALLOONC.EXE by recompiler/recompile.py (regen.sh; not in git).
add_library(rcgame STATIC gen/blocks.cpp gen/image.cpp gen/bootenv.cpp
            ${RC_RUNTIME}/machine.cpp ${RC_RUNTIME}/ega.cpp)
target_include_directories(rcgame PUBLIC ${RC_RUNTIME})
target_compile_options(rcgame PRIVATE -Wno-unused-label)
# Decompiled procedures (src/game): object library so their static registrations are always linked.
file(GLOB RC_GAME_SRC ${CMAKE_CURRENT_SOURCE_DIR}/../src/game/*.cpp)
add_library(rcnative OBJECT ${RC_GAME_SRC})
target_include_directories(rcnative PUBLIC ${RC_RUNTIME})
target_compile_options(rcnative PRIVATE -Wno-unused-label)
add_executable(rc_headless host/headless.cpp host/png.cpp)
target_include_directories(rc_headless PRIVATE host)
target_link_libraries(rc_headless PRIVATE rcgame rcnative)
find_package(SDL2 QUIET)
if(SDL2_FOUND)
  add_executable(rc_play host/sdl_play.cpp host/gamefiles.cpp)
  target_include_directories(rc_play PRIVATE ${SDL2_INCLUDE_DIRS})
  target_link_libraries(rc_play PRIVATE rcgame rcnative ${SDL2_LIBRARIES})
endif()
